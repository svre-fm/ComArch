        add	0	0	1 //รีเซ็ตค่าใน register 1 ให้เป็น 0
        lw	0	2	mcand //โหลดตัวตั้งของการคูณ
        lw	0	3	mplier //โหลดตัวคูณ
        lw	0	4	pos1 //โหลดค่า 1 ที่เป็นบวกสำหรับใช้เป็นตัวเทียบบิต
        lw	0	5	limit //โหลดค่า limit บิตที่เช็คได้ 
        lw	0	6	neg1 //โหลดค่า -1 สำหรับเทียบบิต

        nand	3	4	7 //ทำการ nand reg 3 4 
        beq	7	6	skip //เช็คว่าผลการ nand ว่าบิตนั้นเป็น 0 ไหม (nand = -1) จะทำการข้ามการบวกตัวตั้งที่ถูก shift ไป
        add	1	2	1 //ทำการบวกตัวตั้งเข้าผลลัพธ์

        add	4	4	4 //ทำการ shift บิตตัวเทียบ
        beq	0	0	loop //ทำการวนซ้ำ
